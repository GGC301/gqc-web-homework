五、测试台的核心代码

            

                
async function yuan_request(yuan_method, yuan_url, yuan_body) {
    var yuan_opts = {
        method: yuan_method,
        headers: { 'Accept': 'application/json' }
    };

    // 只有带请求体的方法才设置 body 与 Content-Type
    if (yuan_body && yuan_method !== 'GET') {
        yuan_opts.headers['Content-Type'] = 'application/json';
        yuan_opts.body = yuan_body;          // 必须是字符串，不能直接传对象
    }

    try {
        var yuan_res = await fetch(yuan_url, yuan_opts);

        // 1) HTTP 层面：4xx / 5xx 不会抛错，要自己看 ok
        console.log(yuan_res.status, yuan_res.ok);

        // 2) 先拿文本，避免响应不是 JSON 时解析报错
        var yuan_text = await yuan_res.text();

        var yuan_data = null;
        try { yuan_data = JSON.parse(yuan_text); }
        catch (e) { console.warn('返回的不是 JSON：', yuan_text.slice(0, 50)); }

        if (!yuan_res.ok) {
            throw new Error('HTTP ' + yuan_res.status + ' ' + yuan_res.statusText);
        }
        return yuan_data;
    } catch (yuan_err) {
        // 3) 网络层：断网、域名错误、被 CORS 拦截，都会走到这里
        console.error('请求失败：' + yuan_err.message);
        throw yuan_err;
    }
}
