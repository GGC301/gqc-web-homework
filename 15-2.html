JavaScript 程序结构

条件判断 if / switch · 循环 for / while / do...while · break / continue

一、条件判断

1. if ... else
最基本的条件分支。条件为真执行 if 块，否则执行 else 块。
// 判断一个数是奇数还是偶数 var yuan_num = 7; if (yuan_num % 2 === 0) { yuan_output += yuan_num + " 是偶数\n"; } else { yuan_output += yuan_num + " 是奇数\n"; }
运行结果
2. if ... else if ... else
多分支判断，从上往下依次检查条件，命中即执行。
// 根据分数输出等级 var yuan_score = 85; if (yuan_score >= 90) { yuan_output += yuan_score + " 分 → 优秀\n"; } else if (yuan_score >= 80) { yuan_output += yuan_score + " 分 → 良好\n"; } else if (yuan_score >= 60) { yuan_output += yuan_score + " 分 → 及格\n"; } else { yuan_output += yuan_score + " 分 → 不及格\n"; }
运行结果
3. switch ... case
适合判断"一个值对多种情况"，每个 case 要记得加 break。
// 根据今天是星期几输出安排 var yuan_day = new Date().getDay(); // 0=周日, 1=周一 ... 6=周六 var yuan_plan = ""; switch (yuan_day) { case 1: case 3: case 5: yuan_plan = "上课"; break; case 2: case 4: yuan_plan = "自习"; break; case 6: yuan_plan = "出去玩"; break; case 0: yuan_plan = "睡懒觉"; break; } yuan_output += "今天是周" + ["日","一","二","三","四","五","六"][yuan_day] + "，安排：" + yuan_plan;
运行结果

二、循环

1. for 循环
计数型循环，三部分：初始化；条件；增量。最常用的循环方式。
// 输出 1 到 5 var yuan_text = ""; for (var yuan_i = 1; yuan_i <= 5; yuan_i++) { yuan_text += yuan_i + " "; } yuan_output += "1~5：" + yuan_text.trim() + "\n"; // 求 1 到 100 的和 var yuan_sum = 0; for (var yuan_j = 1; yuan_j <= 100; yuan_j++) { yuan_sum += yuan_j; } yuan_output += "1+2+...+100 = " + yuan_sum;
运行结果
2. while 循环
先判断条件，条件为真就一直执行循环体。
// 计算 2 的 0~4 次幂 var yuan_k = 0; while (yuan_k <= 4) { yuan_output += "2^" + yuan_k + " = " + Math.pow(2, yuan_k) + "\n"; yuan_k++; }
运行结果
3. do ... while 循环
先执行一次循环体，再判断条件。至少执行一次。
// 至少打印一次，然后累加直到和 ≥ 10 var yuan_n = 1; var yuan_total = 0; do { yuan_total += yuan_n; yuan_n++; } while (yuan_total < 10); yuan_output += "累加停止于 n=" + yuan_n + "，总和=" + yuan_total;
运行结果
4. for...of 循环（遍历数组）
ES6 语法，直接取数组里的每个值，比传统 for 更简洁。
// 遍历一个数组 var yuan_fruits = ["苹果", "香蕉", "橙子", "西瓜"]; for (var yuan_fruit of yuan_fruits) { yuan_output += yuan_fruit + " "; }
运行结果

三、循环控制 break / continue

break —— 立刻跳出整个循环
数7规则里，遇到 7 的倍数或含 7 的数就"跳过"，这里演示找到第一个超过 50 且能被 7 整除的数。
for (var yuan_m = 1; yuan_m <= 100; yuan_m++) { if (yuan_m > 50 && yuan_m % 7 === 0) { yuan_output += "第一个 >50 且能被 7 整除的数是：" + yuan_m; break; // 找到后立即跳出循环 } }
运行结果
continue —— 跳过本次循环，进入下一轮
输出 1~20 中所有偶数，遇到奇数直接 continue 不打印。
var yuan_even = ""; for (var yuan_p = 1; yuan_p <= 20; yuan_p++) { if (yuan_p % 2 !== 0) { continue; // 奇数跳过 } yuan_even += yuan_p + " "; } yuan_output += "1~20 的偶数：" + yuan_even.trim();
运行结果
