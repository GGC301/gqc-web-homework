    二、增删元素 会改变原数组

    push() / pop()

    栈（stack）操作：后进先出。push 返回新长度，pop 返回被删除的元素。

    var yuan_stack = ["盘子A", "盘子B"]; yuan_stack.push("盘子C"); // 返回 3 yuan_stack.pop(); // 返回 "盘子C"
    运行结果

    unshift() / shift()

    队列（queue）操作：先进先出。unshift 在头部插入，shift 从头部取出。

    var yuan_queue = ["2号", "3号"]; yuan_queue.unshift("1号"); // 返回 3 yuan_queue.shift(); // 返回 "1号"
    运行结果

    splice() 万能剪刀

    splice(起点, 删几个, 插入…)：可删、可插、可替换，返回被删掉的元素。

    var yuan_cls = ["张三", "李四", "王五", "赵六"]; yuan_cls.splice(1, 2, "关泉才"); // 返回 ["李四","王五"]
    运行结果

    三、截取与拼接 不改变原数组，返回新结果

    slice() 切片

    slice(开始, 结束)：含头不含尾，像切蛋糕；原数组毫发无损。

    var yuan_n = [10, 20, 30, 40, 50]; yuan_n.slice(1, 4); // [20, 30, 40]
    运行结果

    concat() 拼接

    把多个数组 / 值连成一条新数组，相当于"车厢对接"。

    var yuan_a = ["车头"]; var yuan_b = ["车厢1", "车厢2"]; yuan_a.concat(yuan_b, "车尾");
    运行结果

    join() / toString()

    把数组挤成一根字符串，join 可以指定"胶水"。

    var yuan_p = ["北京", "上海", "广州"]; yuan_p.join("-"); // "北京-上海-广州" yuan_p.toString(); // "北京,上海,广州"
    运行结果

    四、查找与判断 indexOf / includes / find

    indexOf() / lastIndexOf()

    按图索骥：找到返回下标，找不到返回 -1。

    var yuan_c = ["红", "绿", "蓝", "绿"]; yuan_c.indexOf("绿"); // 1 yuan_c.lastIndexOf("绿"); // 3 yuan_c.indexOf("紫"); // -1
    运行结果

    includes()

    只回答"在不在"：返回 true / false，比 indexOf 更直观。

    var yuan_menu = ["米饭", "面条", "饺子"]; yuan_menu.includes("饺子"); // true yuan_menu.includes("披萨"); // false
    运行结果

    find() / findIndex()

    按条件（回调函数）找：find 返回元素本身，findIndex 返回下标。

    var yuan_s = [55, 72, 91, 68]; yuan_s.find(function (v) { return v >= 90; }); // 91 yuan_s.findIndex(function (v) { return v >= 90; }); // 2
    运行结果

    五、遍历与加工 forEach / map / filter / reduce

    forEach() 逐个点名

    每个元素都执行一次回调，不返回新数组，适合"做事"。

    var yuan_t = ["早读", "上课", "午休"]; yuan_t.forEach(function (v, i) { console.log(i + "：" + v); });
    运行结果

    map() 加工流水线

    一对一变形：进去几个出来几个，返回全新数组。

    var yuan_price = [10, 20, 30]; yuan_price.map(function (v) { return v * 2; }); // [20,40,60]
    运行结果

    filter() 筛子

    留下条件为 true 的元素，数组长度可能变短。

    var yuan_age = [12, 18, 25, 9, 30]; yuan_age.filter(function (v) { return v >= 18; }); // [18,25,30]
    运行结果

    reduce() 压成一个值

    带着"累计器"从左到右滚过去，求和、求积、统计都靠它。

    var yuan_m = [80, 90, 100]; yuan_m.reduce(function (sum, v) { return sum + v; }, 0); // 270
    运行结果

    every() / some()

    every 要求"全都满足"，some 只要"有一个满足"。

    var yuan_sc = [70, 85, 92]; yuan_sc.every(function (v) { return v >= 60; }); // 全及格 → true yuan_sc.some(function (v) { return v >= 90; }); // 有优秀 → true
    运行结果

    flat() 压平

    把嵌套数组"拍扁"一层（或多层），非常好用。

    var yuan_nest = [1, [2, 3], [4, [5]]]; yuan_nest.flat(); // [1,2,3,4,[5]] yuan_nest.flat(2); // [1,2,3,4,5]
    运行结果

    六、排序与其他常用方法

    sort() 排序（要写比较函数）

    默认按字符串排序，数字排序必须传 (a, b) => a - b。

    var yuan_num = [10, 2, 33, 4]; yuan_num.sort(); // 错误示范：[10,2,33,4] yuan_num.sort(function (a, b) { return a - b; }); // [2,4,10,33]
    运行结果

    reverse() 反转

    把数组掉个头，原地修改。

    var yuan_line = ["排头", "中间", "排尾"]; yuan_line.reverse(); // ["排尾","中间","排头"]
    运行结果

    Array.isArray() / Array.from()

    判断是不是数组；把类数组（如字符串）转成真数组。

    Array.isArray([1, 2]); // true Array.isArray("abc"); // false Array.from("abc"); // ["a","b","c"]
    运行结果

    fill() / length

    fill 批量填充；length 既能读也能写（写小了会截断）。

    var yuan_f = [1, 2, 3, 4]; yuan_f.fill(0); // [0,0,0,0] yuan_f.length = 2; // [0,0]
    运行结果

    七、方法速查表

    方法 作用（形象说法） 改变原数组 返回值
    push()队尾上车是新长度
    pop()队尾下车是被删元素
    unshift()队头插队是新长度
    shift()队头放行是被删元素
    splice()万能剪刀：删 / 插 / 换是被删元素数组
    sort()重新排座位是排序后的原数组
    reverse()整队掉头是反转后的原数组
    fill()批量灌同样的值是填充后的原数组
    slice()切一片蛋糕否新数组
    concat()车厢对接否新数组
    join()用胶水粘成字符串否字符串
    indexOf()按图索骥否下标 / -1
    includes()在不在？否true / false
    find() / findIndex()按条件挑人否元素 / 下标
    forEach()逐个点名做事否undefined
    map()加工流水线否新数组
    filter()筛子过滤否新数组
    reduce()滚雪球成一个值否累计结果
    every() / some()全满足 / 有满足否true / false
    flat()把嵌套拍扁否新数组
