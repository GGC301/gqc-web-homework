// 1) 从地址栏取 id：17-3-detail.html?id=3 var yuan_id = Number(new URLSearchParams(location.search).get('id')) || 1; // 2) 用 id 拼接口地址 var yuan_url = 'https://dummyjson.com/products/' + yuan_id; // 3) 请求 async function yuan_loadDetail() { var yuan_res = await fetch(yuan_url); if (!yuan_res.ok) { // 404 也要自己判断 throw new Error('HTTP ' + yuan_res.status); } var yuan_item = await yuan_res.json(); document.getElementById('yuan_title').textContent = yuan_item.title; // images 是数组，可以做成缩略图切换 console.log(yuan_item.images.length); } // 4) 换商品 = 改地址栏参数后重新请求 function yuan_go(yuan_newId) { location.href = '17-3-detail.html?id=' + yuan_newId; }
